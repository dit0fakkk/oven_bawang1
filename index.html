// ─── Google Apps Script ─────────────────────────
// Extensions → Apps Script → paste ini → Run

const FIREBASE_URL =
  "https://your-project-rtdb.firebaseio.com";
const PROJECT_KEY  = "oven_bawang";

function importFromFirebase() {
  const url = `${FIREBASE_URL}/${PROJECT_KEY}/log.json`;
  const res  = UrlFetchApp.fetch(url);
  const data = JSON.parse(res.getContentText());

  const sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName("Data Log")
    || SpreadsheetApp.getActiveSpreadsheet()
         .insertSheet("Data Log");

  // Header
  sheet.getRange(1,1,1,6).setValues([[
    "Waktu","Suhu (°C)","Kelembaban (%)","Fan","Lampu","Status"
  ]]);

  const rows = Object.values(data || {});
  rows.sort((a,b) => a.ts - b.ts);

  rows.forEach((d, i) => {
    sheet.getRange(i+2,1,1,6).setValues([[
      new Date(d.ts),
      d.suhu, d.humi,
      d.fan ? "ON" : "OFF",
      d.lampu ? "ON" : "OFF",
      d.suhu >= 60 ? "PANAS" : "NORMAL"
    ]]);
  });

  SpreadsheetApp.getUi().alert(
    `✅ Import ${rows.length} baris selesai!`
  );
}

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu("🧅 Oven Bawang")
    .addItem("📥 Import dari Firebase","importFromFirebase")
    .addToUi();
}
